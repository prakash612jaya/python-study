import random # importing module to use random numbers
import string # importing module to get letters

# creating random numbers from 2 to 10
r1 = []
rand_num = random.randint(2,10) # pick a random number of dicts to generate
for i in range(rand_num):
    curr_dict = {}
    num_keys = random.randint(1,5) # pick a random number of keys for this dict
    keys = random.sample(string.ascii_lowercase, num_keys) # pick random letters for keys
    for key in keys:
        curr_dict[key] = random.randint(0,100)
        r1.append(curr_dict)
# print the generated dictionaries
print("Generated list of dictionaries:")
for i, d in enumerate(r1):
    print(f"Dic {i + 1}:", d)

# creating one common dictionary
merged_dict = {} # this will hold the final result
key_origins = {} # this keeps track of which dict had the max value for each key

for i, d in enumerate(r1):
    for key, value in d.items():
        if key in merged_dict: # if key already exists, compare values
            if value > merged_dict[key]:
                merged_dict[key] = value # update to bigger value
                key_origins[key] = i + 1 # remember which dict has max value
        else:
            merged_dict[key] = value # first time seeing this key
            key_origins[key] = i + 1 # save which dict this key came from
final_dict = {}
for key, value in merged_dict.items():
    count = sum(key in d for d in r1) # count how many times this key appeared
    if count > 1: # if key appeared more than once, rename it with the dict number that has max value
        new_key = f"{key}_{key_origins[key]}"
        final_dict[new_key] = value
    else:
        final_dict[key] = value

print("\nFinal merged dictionary:")
print(final_dict)
